cmake_minimum_required(VERSION 3.15)
project(TutorialNeuroSymbolic VERSION 0.1.0)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(AriadneNeuroSymbolic REQUIRED)

add_executable(tutorial_neurosymbolic tutorial_neurosymbolic.cpp)
target_link_libraries(tutorial_neurosymbolic PRIVATE
    AriadneNeuroSymbolic::ariadne-neuro-symbolic)
