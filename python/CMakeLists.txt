set(Python3_FIND_VIRTUALENV FIRST)
set(Python3_FIND_STRATEGY LOCATION)
set(PYBIND11_FINDPYTHON ON)

find_package(Python3 3.11 COMPONENTS Interpreter Development.Module)
option(PYTHON_ENFORCE_TESTS "Enforce Python tests" OFF)

if(Python3_FOUND)
    report_python_configuration_once()
    if(NOT TARGET pyariadne-neural OR NOT TARGET pyariadne-expression_set)
        message(FATAL_ERROR "Neural and Expression Set Python binding targets are required.")
    endif()
    if(NOT TARGET pyariadne-python-common OR NOT TARGET pybind11::module)
        message(FATAL_ERROR "PythonCommon and pybind11 are required.")
    endif()
    add_subdirectory(bindings)

    if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
        ariadne_finalize_python_module(NAME neurosymbolic MODULE_IMPORT_NAME pyariadne_neurosymbolic)
        if(WIN32)
            set(_python_install_dir "Lib/site-packages")
        else()
            set(_python_install_dir "lib/python${Python3_VERSION_MAJOR}.${Python3_VERSION_MINOR}/site-packages")
        endif()
        set(PYTHON_BINDINGS_INSTALL_DIR "${_python_install_dir}"
            CACHE STRING "Install destination relative to the prefix")
        install(TARGETS pyariadne DESTINATION "${PYTHON_BINDINGS_INSTALL_DIR}")
        add_subdirectory(tests)
        add_dependencies(bindings pyariadne)
    endif()
else()
    if(PYTHON_ENFORCE_TESTS)
        message(FATAL_ERROR "Python >= 3.11 with Interpreter and Development.Module is required.")
    else()
        message(STATUS "Compatible Python development environment unavailable; bindings disabled.")
    endif()
endif()
